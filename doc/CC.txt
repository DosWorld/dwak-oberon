Conditional compilation

syntax:

    $IF "(" ident {"|" ident} ")"
        <...>
    {$ELSIF "(" ident {"|" ident} ")"}
        <...>
    [$ELSE]
        <...>
    $END

    where ident is:
    - one of the possible values of the <target> parameter on the command line
    - a user-defined identifier passed with the -def option at compile time
    - one of the possible predefined identifiers:

        WINDOWS   - Windows application
        LINUX     - Linux application
        KOLIBRIOS - KolibriOS application
        CPU_X86   - application for the x86 (32-bit) processor
        CPU_X8664 - application for the x86_64 processor


examples:

    $IF (win64con | win64gui | win64dll)
        OS := "WIN64";
    $ELSIF (win32con | win32gui | win32dll)
        OS := "WIN32";
    $ELSIF (linux64exe | linux64so)
        OS := "LINUX64";
    $ELSIF (linux32exe | linux32so)
        OS := "LINUX32";
    $ELSE
        OS := "UNKNOWN";
    $END


    $IF (debug) (* -def debug *)
        print("debug");
    $END


    $IF (WINDOWS)
        $IF (CPU_X86)
        (*windows 32*)

        $ELSIF (CPU_X8664)
        (*windows 64*)

        $END
    $ELSIF (LINUX)
        $IF (CPU_X86)
        (*linux 32*)

        $ELSIF (CPU_X8664)
        (*linux 64*)

        $END
    $END
