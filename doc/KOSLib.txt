MODULE Out - console output

        PROCEDURE Open
                formally opens the console output

        PROCEDURE Int(x, width: INTEGER)
                output of the integer x;
                width - the number of character positions used for the output

        PROCEDURE Real(x: REAL; width: INTEGER)
                output of the real number x in floating-point format;
                width - the number of character positions used for the output

        PROCEDURE Char(x: CHAR)
                output of the character x

        PROCEDURE FixReal(x: REAL; width, p: INTEGER)
                output of the real number x in fixed-point format;
                width - the number of character positions used for the output;
                p - the number of digits after the decimal point

        PROCEDURE Ln
                move to the next line

        PROCEDURE String(s: ARRAY OF CHAR)
                output of the string s

MODULE File - file system operations

        TYPE

                FNAME = ARRAY 520 OF CHAR

                FS = POINTER TO rFS

                rFS = RECORD (* file information structure *)
                        subfunc, pos, hpos, bytes, buffer: INTEGER;
                        name: FNAME
                END

                FD = POINTER TO rFD

                rFD = RECORD (* structure of a directory entry data block *)
                        attr: INTEGER;
                        ntyp: CHAR;
                        reserved: ARRAY 3 OF CHAR;
                        time_create, date_create,
                        time_access, date_access,
                        time_modif,  date_modif,
                        size, hsize: INTEGER;
                        name: FNAME
                END

        CONST

                SEEK_BEG = 0
                SEEK_CUR = 1
                SEEK_END = 2

        PROCEDURE Load(FName: ARRAY OF CHAR; VAR size: INTEGER): INTEGER;
                Loads the file named FName into memory, writes the file size
                into the parameter size, returns the address of the loaded file
                or 0 (error). If necessary, the file is unpacked
                (kunpack).

        PROCEDURE GetFileInfo(FName: ARRAY OF CHAR; VAR Info: rFD): BOOLEAN
                Writes the directory entry data block structure for the file
                or folder named FName into the parameter Info.
                Returns FALSE on error.

        PROCEDURE Exists(FName: ARRAY OF CHAR): BOOLEAN
                returns TRUE if the file named FName exists

        PROCEDURE Close(VAR F: FS)
                frees the memory allocated for the information structure of
                file F and assigns NIL to F

        PROCEDURE Open(FName: ARRAY OF CHAR): FS
                returns a pointer to the information structure of the file
                named FName, returns NIL on error

        PROCEDURE Delete(FName: ARRAY OF CHAR): BOOLEAN
                deletes the file named FName, returns FALSE on error

        PROCEDURE Seek(F: FS; Offset, Origin: INTEGER): INTEGER
                sets the read-write position of file F to Offset,
                relative to Origin = (SEEK_BEG - beginning of file,
                SEEK_CUR - current position, SEEK_END - end of file),
                returns the position relative to the beginning of the file, for example:
                        Seek(F, 0, SEEK_END)
                sets the position to the end of the file and returns the
                file length; returns -1 on error

        PROCEDURE Read(F: FS; Buffer, Count: INTEGER): INTEGER
                Reads data from the file into memory. F - pointer to the
                information structure of the file, Buffer - address of the
                memory area, Count - the number of bytes to be read from the
                file; returns the number of bytes that were read and changes
                the read/write position in the information structure F
                accordingly.

        PROCEDURE Write(F: FS; Buffer, Count: INTEGER): INTEGER
                Writes data from memory to the file. F - pointer to the
                information structure of the file, Buffer - address of the
                memory area, Count - the number of bytes to be written to the
                file; returns the number of bytes that were written and
                changes the read/write position in the information structure
                F accordingly.

        PROCEDURE Create(FName: ARRAY OF CHAR): FS
                creates a new file named FName (full name), returns a pointer
                to the information structure of the file,
                returns NIL on error

        PROCEDURE CreateDir(DirName: ARRAY OF CHAR): BOOLEAN
                creates the folder named DirName, all intermediate folders
                must exist, returns FALSE on error

        PROCEDURE DeleteDir(DirName: ARRAY OF CHAR): BOOLEAN
                deletes the empty folder named DirName,
                returns FALSE on error

        PROCEDURE DirExists(DirName: ARRAY OF CHAR): BOOLEAN
                returns TRUE if the folder named DirName exists

MODULE KOSAPI

        PROCEDURE sysfunc1(arg1: INTEGER): INTEGER
        PROCEDURE sysfunc2(arg1, arg2: INTEGER): INTEGER
        ...
        PROCEDURE sysfunc7(arg1, arg2, ..., arg7: INTEGER): INTEGER
                Wrappers for KolibriOS kernel API functions.
                arg1 .. arg7 correspond to the registers
                        eax, ebx, ecx, edx, esi, edi, ebp;
                they return the value of the eax register after the system call.

        PROCEDURE sysfunc22(arg1, arg2: INTEGER; VAR res2: INTEGER): INTEGER
                Wrapper for KolibriOS kernel API functions.
                arg1 - the eax register, arg2 - the ebx register,
                res2 - the value of the ebx register after the system call;
                returns the value of the eax register after the system call.

        PROCEDURE malloc(size: INTEGER): INTEGER
                Allocates a memory block.
                size - the size of the block in bytes,
                returns the address of the allocated block

        PROCEDURE free(ptr: INTEGER): INTEGER
                Frees a previously allocated memory block with the address ptr,
                returns 0

        PROCEDURE realloc(ptr, size: INTEGER): INTEGER
                Reallocates a memory block,
                ptr - the address of the previously allocated block,
                size - the new size,
                returns a pointer to the reallocated block,
                0 on error

        PROCEDURE GetCommandLine(): INTEGER
                Returns the address of the parameter string

        PROCEDURE GetName(): INTEGER
                Returns the address of the string with the program name

        PROCEDURE LoadLib(name: ARRAY OF CHAR): INTEGER
                Loads a DLL with the full name name. Returns the address of the
                export table. Returns 0 on error.

        PROCEDURE GetProcAdr(name: ARRAY OF CHAR; lib: INTEGER): INTEGER
                name - the procedure name
                lib - the address of the DLL export table
                Returns the address of the procedure. Returns 0 on error.

                                         1 - the user pressed OK
                                         2 - the dialog is open  *)

                    color:  INTEGER   (* the selected colour *)
                END

        PROCEDURE Create(draw_window: DRAW_WINDOW): Dialog
                create a dialog

                (TYPE DRAW_WINDOW = PROCEDURE);

        PROCEDURE Show(cd: Dialog)
                show the dialog
                cd - pointer to the dialog structure created earlier by the
                procedure Create

        PROCEDURE Destroy(VAR cd: Dialog)

MODULE OpenDlg - working with the "Open Dialog" dialog

        TYPE

                Dialog = POINTER TO RECORD (* dialog structure *)
                    status:   INTEGER  (* dialog state:
                                          0 - the user pressed Cancel
                                          1 - the user pressed OK
                                          2 - the dialog is open  *)

                    FileName: ARRAY 4096 OF CHAR (* name of the selected file *)
                    FilePath: ARRAY 4096 OF CHAR (* full name of the selected
                                                    file            *)
                END

        PROCEDURE Create(draw_window: DRAW_WINDOW; type: INTEGER; def_path,
                        filter: ARRAY OF CHAR): Dialog
                create a dialog
                draw_window - procedure for redrawing the main window
                        (TYPE DRAW_WINDOW = PROCEDURE)
                type -  dialog type
                        0 - open
                        1 - save
                        2 - select a folder
                def_path - default path, the folder def_path will be opened
                        when the dialog is started for the first time
                filter - the string contains a list of file extensions that
                        will be shown in the dialog window, the extensions are
                        separated by the "|" character, e.g. "ASM|TXT|INI"
                the procedure returns a pointer to the dialog structure

        PROCEDURE Show(od: Dialog; Width, Height: INTEGER)
                show the dialog
                od - pointer to the dialog structure created earlier by the
                     procedure Create
                Width and Height - width and height of the dialog window

        PROCEDURE Destroy(VAR od: Dialog)
                destroy the dialog
                od - pointer to the dialog structure

MODULE kfonts - working with kf fonts

        CONST

                bold            =   1
                italic          =   2
                underline       =   4
                strike_through  =   8
                smoothing       =  16
                bpp32           =  32

        TYPE

                TFont = POINTER TO TFont_desc (* pointer to a font *)

        PROCEDURE LoadFont(file_name: ARRAY OF CHAR): TFont
                load a font from a file
                file_name   name of the kf file
                result:         pointer to the font/NIL (error)

        PROCEDURE SetSize(Font: TFont; font_size: INTEGER): BOOLEAN
                set the font size
                Font        pointer to the font
                font_size   font size
                result:         TRUE/FALSE (error)

        PROCEDURE Enabled(Font: TFont; font_size: INTEGER): BOOLEAN
                check whether a font of the given size is present
                Font        pointer to the font
                font_size   font size
                result:         TRUE/FALSE (the font is not present)

        PROCEDURE Destroy(VAR Font: TFont)
                unload the font, free the dynamic memory
                Font        pointer to the font
                Assigns NIL to the variable Font

        PROCEDURE TextHeight(Font: TFont): INTEGER
                get the height of a text line
                Font        pointer to the font
                result:         height of the text line in pixels

        PROCEDURE TextWidth(Font: TFont;
                    str, length, params: INTEGER): INTEGER
                get the width of a text line
                Font        pointer to the font
                str         address of the text string in Win-1251 encoding
                length      number of characters in the string, or -1 if the string
                            is terminated by zero
                params      flag parameters, see below
                result:         width of the text line in pixels

        PROCEDURE TextOut(Font: TFont;
                    canvas, x, y, str, length, color, params: INTEGER)
                output text into a buffer
                to output the buffer into a window, use function 65 or
                function 7 (if the buffer is 24-bit)
                Font        pointer to the font
                canvas      address of the graphics buffer
                            buffer structure:
                              Xsize       dd
                              Ysize       dd
                              picture     rb  Xsize * Ysize * 4 (32 bits)
                                              or Xsize * Ysize * 3 (24 bits)
                x, y        coordinates of the text relative to the top left
                            corner of the buffer
                str         address of the text string in Win-1251 encoding
                length      number of characters in the string, or -1 if the string
                            is terminated by zero
                color       text colour 0x00RRGGBB
                params      flag parameters:
                               1   bold
                               2   italic
                               4   underlined
                               8   strike-through
                              16   apply smoothing
                              32   output to a 32-bit buffer
                            the flags may be used in any combination
