        Library (Windows)

MODULE Out - console output

        PROCEDURE Open
                opens console output

        PROCEDURE Int(x, width: INTEGER)
                outputs the integer x;
                width - the number of character positions used for the output

        PROCEDURE Real(x: REAL; width: INTEGER)
                outputs the real number x in floating-point format;
                width - the number of character positions used for the output

        PROCEDURE Char(x: CHAR)
                outputs the character x

        PROCEDURE FixReal(x: REAL; width, p: INTEGER)
                outputs the real number x in fixed-point format;
                width - the number of character positions used for the output;
                p - the number of digits after the decimal point

        PROCEDURE Ln
                moves to the next line

        PROCEDURE String(s: ARRAY OF CHAR)
                outputs the string s (ASCII)

        PROCEDURE StringW(s: ARRAY OF WCHAR)
                outputs the string s (UTF-16)

------------------------------------------------------------------------------
MODULE In - console input

        VAR Done: BOOLEAN
                is set to TRUE when the input operation completes
                successfully and to FALSE otherwise

        PROCEDURE Open
                opens console input;
                also sets the variable Done to TRUE

        PROCEDURE Int(VAR x: INTEGER)
                reads an INTEGER value

        PROCEDURE Char(VAR x: CHAR)
                reads a character

        PROCEDURE Real(VAR x: REAL)
                reads a REAL value

        PROCEDURE String(VAR s: ARRAY OF CHAR)
                reads a string

        PROCEDURE Ln
                waits for ENTER to be pressed

MODULE Console - additional console output procedures

        CONST

        The following constants define the console output colour

                Black = 0      Blue = 1           Green = 2
                Cyan = 3       Red = 4            Magenta = 5
                Brown = 6      LightGray = 7      DarkGray = 8
                LightBlue = 9  LightGreen = 10    LightCyan = 11
                LightRed = 12  LightMagenta = 13  Yellow = 14
                White = 15

        PROCEDURE Cls
                clears the console window

        PROCEDURE SetColor(FColor, BColor: INTEGER)
                sets the colour of console output: FColor - text colour,
                BColor - background colour; the possible values are the
                constants listed above

        PROCEDURE SetCursor(x, y: INTEGER)
                sets the console cursor to position (x, y)

        PROCEDURE GetCursor(VAR x, y: INTEGER)
                writes the current console cursor coordinates into the parameters

        PROCEDURE GetCursorX(): INTEGER
                returns the current x coordinate of the console cursor

        PROCEDURE GetCursorY(): INTEGER
                returns the current y coordinate of the console cursor

MODULE File - file system operations

        CONST

                OPEN_R = 0
                OPEN_W = 1
                OPEN_RW = 2

                SEEK_BEG = 0
                SEEK_CUR = 1
                SEEK_END = 2

        PROCEDURE Create(FName: ARRAY OF CHAR): INTEGER
                creates a new file named FName (full name including the path),
                opens the file for writing and returns the file identifier
                (an integer); on error returns -1

        PROCEDURE Open(FName: ARRAY OF CHAR; Mode: INTEGER): INTEGER
                opens the existing file named FName (full name including
                the path) in the mode Mode = (OPEN_R (read only), OPEN_W
                (write only), OPEN_RW (read and write)); returns
                the file identifier (an integer); on error
                returns -1

        PROCEDURE Read(F, Buffer, Count: INTEGER): INTEGER
                Reads data from a file into memory. F - the numeric file
                identifier, Buffer - the address of the memory area, Count - the number of bytes
                to read from the file; returns the number of bytes
                actually read from the file

        PROCEDURE Write(F, Buffer, Count: INTEGER): INTEGER
                Writes data from memory to a file. F - the numeric file
                identifier, Buffer - the address of the memory area, Count - the number of bytes
                to write to the file; returns the number of bytes
                actually written to the file

        PROCEDURE Seek(F, Offset, Origin: INTEGER): INTEGER
                sets the read-write position of the file with identifier F
                to Offset, relative to Origin = (SEEK_BEG - beginning of the file,
                SEEK_CUR - current position, SEEK_END - end of the file),
                returns the position relative to the beginning of the file, for example:
                Seek(F, 0, 2) sets the position to the end of the file and
                returns the length of the file; on error returns -1

        PROCEDURE Close(F: INTEGER)
                closes the previously opened file with identifier F

        PROCEDURE Delete(FName: ARRAY OF CHAR): BOOLEAN
                deletes the file named FName (full name including the path);
                returns TRUE if the file was deleted successfully

        PROCEDURE Exists(FName: ARRAY OF CHAR): BOOLEAN
                returns TRUE if a file named FName (full name)
                exists

        PROCEDURE Load(FName: ARRAY OF CHAR; VAR Size: INTEGER): INTEGER
                loads the existing file named FName (full name including
                the path) into memory; returns the memory address where the file was loaded,
                writes the file size into the parameter Size;
                on error returns 0

        PROCEDURE CreateDir(DirName: ARRAY OF CHAR): BOOLEAN
                creates a directory named DirName; all intermediate directories
                must already exist. On error returns FALSE

        PROCEDURE RemoveDir(DirName: ARRAY OF CHAR): BOOLEAN
                removes the empty directory named DirName. On error
                returns FALSE

        PROCEDURE ExistsDir(DirName: ARRAY OF CHAR): BOOLEAN
                returns TRUE if a directory named DirName exists

MODULE DateTime - date and time

        CONST ERR = -7.0E5

        PROCEDURE Now(VAR Year, Month, Day, Hour, Min, Sec, MSec: INTEGER)
                returns the components of the current system date and
                time in the parameters

        PROCEDURE NowEncode(): REAL;
                returns the current system date and
                time (REAL representation)

        PROCEDURE Encode(Year, Month, Day,
            Hour, Min, Sec, MSec: INTEGER): REAL
                returns the date built from the components
                Year, Month, Day, Hour, Min, Sec, MSec;
                on error returns the constant ERR = -7.0E5

        PROCEDURE Decode(Date: REAL; VAR Year, Month, Day,
                Hour, Min, Sec, MSec: INTEGER): BOOLEAN
                extracts the components
                Year, Month, Day, Hour, Min, Sec, MSec from the date Date;
                on error returns FALSE

MODULE Args - program parameters

        VAR argc: INTEGER
                the number of program parameters, including
                the name of the executable file

        PROCEDURE GetArg(n: INTEGER; VAR s: ARRAY OF CHAR)
                writes the n-th program parameter into the string s;
                parameters are numbered from 0 to argc - 1,
                parameter zero is the name of the executable file

MODULE Utils - miscellaneous

        PROCEDURE Utf8To16(source: ARRAY OF CHAR;
            VAR dest: ARRAY OF CHAR): INTEGER;
                converts the characters of the string source from UTF-8 to
                UTF-16, writes the result into the string dest and
                returns the number of 16-bit characters written into dest

        PROCEDURE PutSeed(seed: INTEGER)
                Initializes the random number generator with the integer seed

        PROCEDURE Rnd(range: INTEGER): INTEGER
                Integer random numbers in the range 0 <= x < range
